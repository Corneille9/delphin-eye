# Runtime dependencies of the Delphin Eye desktop app.
# Training / notebook tooling lives in requirements-dev.txt and is never packaged.

# CPU-only PyTorch: inference runs on CPU (see PredictionService), so the CUDA
# wheels would add ~4 GB to the bundle for nothing. macOS wheels are CPU-only already.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.11.0+cpu; sys_platform != "darwin"
torchvision==0.26.0+cpu; sys_platform != "darwin"
torch==2.11.0; sys_platform == "darwin"
torchvision==0.26.0; sys_platform == "darwin"

ultralytics
sahi
pillow

nicegui
pywebview
